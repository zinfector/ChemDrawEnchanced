cmake_minimum_required(VERSION 3.24)
project(ChemDrawNativePatcher VERSION 1.0.95 LANGUAGES C CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_TESTING OFF CACHE BOOL "Build tests" FORCE)
add_subdirectory(runtime)
add_executable(ChemDrawPatchManager WIN32 main.cpp "${CMAKE_CURRENT_BINARY_DIR}/payload-$<CONFIG>.rc")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/payload-$<CONFIG>.rc" CONTENT
"#include <windows.h>\n101 RCDATA \"${CMAKE_CURRENT_SOURCE_DIR}/runtime/bin/$<CONFIG>/ChemDrawLatency.dll\"\n102 RCDATA \"${CMAKE_CURRENT_SOURCE_DIR}/runtime/vendor/minhook/LICENSE.txt\"\n201 ICON \"${CMAKE_CURRENT_SOURCE_DIR}/ChemDrawPatchManager.ico\"\n1 RT_MANIFEST \"${CMAKE_CURRENT_SOURCE_DIR}/patcher.manifest\"\n")
add_dependencies(ChemDrawPatchManager ChemDrawLatency)
target_compile_definitions(ChemDrawPatchManager PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(ChemDrawPatchManager PRIVATE /W4 /EHsc /O2 /utf-8)
target_link_options(ChemDrawPatchManager PRIVATE /MANIFEST:NO)
target_link_libraries(ChemDrawPatchManager PRIVATE bcrypt user32 comctl32 shell32 ole32 uuid)
set_target_properties(ChemDrawPatchManager PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/release")
